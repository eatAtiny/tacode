# Shell 沙箱 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 shell 工具加可开关的沙箱（网络隔离 + 文件系统隔离 + 资源限制 + 工作目录隔离），`-sandbox on` 显式启用，默认关闭向后兼容。

**Architecture:** `internal/sandbox` 包定义 `Sandbox` 接口 + `NewSandbox` 工厂 + 平台后端（macOS sandbox-exec / Linux Landlock+unshare / Noop）。`ShellTool` 加 `sandbox` 字段 + `NewShellToolWithSandbox`，`Execute` 时 `Wrap(cmd)`。`main.go` 加 `-sandbox` flag。

**Tech Stack:** Go 1.26.4。macOS 用系统 `sandbox-exec`（无新依赖）；Linux 用 `github.com/landlock-lsm/go-landlock`（仅 linux build tag 拉取）。

// Package sandbox 提供命令沙箱抽象：网络隔离、文件系统隔离、资源限制。
//
// 设计（路径 A）：接口抽象 + 平台原生后端。
//   - macOS:  sandbox-exec（seatbelt profile，内置、非 root 可用）
//   - Linux:  Landlock（文件系统，纯 syscall）+ unshare -n（网络，空网络命名空间）
//   - 其他:   NoopSandbox（无隔离，警告降级）
//
// 沙箱是防御层（纵深防御第二层），默认关闭，用户显式 -sandbox on 启用。
package sandbox

import (
	"fmt"
	"os/exec"
	"runtime"
)

// Sandbox 定义命令沙箱约束（网络/文件系统/资源）。
type Sandbox interface {
	// Wrap 包装 exec.Cmd，注入沙箱约束。返回新 cmd（或原地修改后返回）。
	Wrap(cmd *exec.Cmd) *exec.Cmd
	// AllowsNetwork 报告该沙箱是否允许网络（供 shell 工具判权限）。
	AllowsNetwork() bool
	// Close 释放资源（如临时 profile 文件）。
	Close() error
}

// Config 沙箱配置。
type Config struct {
	AllowNetwork bool   // 是否允许网络（默认 false，防外发）
	WorkDir      string // 可写目录（工作目录，通常为 cwd）
}

// NewSandbox 按当前平台创建沙箱后端。
//
// 平台不支持时返回 NoopSandbox（无隔离）——不阻断启动，调用方可检查
// sandbox.IsActive() 决定是否提示用户。
func NewSandbox(cfg Config) Sandbox {
	switch runtime.GOOS {
	case "darwin":
		return newMacOSSandbox(cfg)
	case "linux":
		return newLinuxSandbox(cfg)
	default:
		return &NoopSandbox{}
	}
}

// NoopSandbox 无隔离后端（平台不支持时降级）。
type NoopSandbox struct{}

func (s *NoopSandbox) Wrap(cmd *exec.Cmd) *exec.Cmd { return cmd }
func (s *NoopSandbox) AllowsNetwork() bool          { return true }
func (s *NoopSandbox) Close() error                 { return nil }

// IsActive 报告沙箱是否真正生效（非 Noop）。
func IsActive(s Sandbox) bool {
	_, ok := s.(*NoopSandbox)
	return !ok
}

// ──────────────────────────────────────────────────────────
// 工厂辅助：平台构建分派（避免 go:build 污染主文件）
// ──────────────────────────────────────────────────────────

// newMacOSSandbox 由 macos.go 实现（//go:build darwin）。
// newLinuxSandbox 由 linux.go 实现（//go:build linux）。
var (
	_ = fmt.Sprintf // 保留 fmt 引用（平台文件使用）
)
```

//go:build darwin

package sandbox

import (
	"fmt"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

// MacOSSandbox 用 sandbox-exec（seatbelt profile）包装命令。
//
// sandbox-exec 虽被 Apple 标记 deprecated，但仍是内置、非 root 可用的
// 最实用选项：一条 profile 同时做到拒绝网络 + 限制文件系统写路径。
type MacOSSandbox struct {
	profilePath  string // 临时 seatbelt profile 文件路径
	allowNetwork bool
	workDir      string // 可写目录
}

// newMacOSSandbox 创建 macOS 沙箱，生成临时 profile 文件。
func newMacOSSandbox(cfg Config) *MacOSSandbox {
	s := &MacOSSandbox{
		allowNetwork: cfg.AllowNetwork,
		workDir:      cfg.WorkDir,
	}
	if s.workDir == "" {
		s.workDir, _ = os.Getwd()
	}
	// 生成临时 profile（Close 时删除）。
	if f, err := os.CreateTemp("", "agentic-sandbox-*.sb"); err == nil {
		fmt.Fprint(f, s.buildProfile())
		f.Close()
		s.profilePath = f.Name()
	}
	return s
}

// buildProfile 生成 seatbelt profile（Lisp 风格，规则有优先级语义）。
func (s *MacOSSandbox) buildProfile() string {
	var sb strings.Builder
	sb.WriteString("(version 1)\n")
	sb.WriteString("(deny default)\n")
	sb.WriteString("(import \"system.sb\")\n")
	// 文件系统：读全放行（读安全），写仅限工作目录。
	sb.WriteString("(allow file-read*)\n")
	sb.WriteString(fmt.Sprintf("(allow file-write* (subpath \"%s\"))\n", s.workDir))
	// 进程：放行进程创建（shell 需要 fork 子进程）。
	sb.WriteString("(allow process*)\n")
	if s.allowNetwork {
		// 网络放行（默认 deny default 已拦，这里显式允许）。
		sb.WriteString("(allow network*)\n")
	} else {
		// 网络全拒（防外发）。
		sb.WriteString("(deny network*)\n")
	}
	return sb.String()
}

// Wrap 用 sandbox-exec 包装命令。
// 最终命令: sandbox-exec -f <profile> -- <cmd.Args...>
func (s *MacOSSandbox) Wrap(cmd *exec.Cmd) *exec.Cmd {
	if s.profilePath == "" {
		return cmd // profile 生成失败 → 降级无沙箱
	}
	inner := append([]string{"-f", s.profilePath, "--"}, cmd.Args...)
	cmd.Args = append([]string{"sandbox-exec"}, inner...)
	cmd.Path = "sandbox-exec"
	return cmd
}

// AllowsNetwork 报告是否允许网络。
func (s *MacOSSandbox) AllowsNetwork() bool { return s.allowNetwork }

// Close 删除临时 profile 文件。
func (s *MacOSSandbox) Close() error {
	if s.profilePath != "" {
		return os.Remove(s.profilePath)
	}
	return nil
}

// 确保沙箱在 darwin 下可用（编译期断言）。
var _ Sandbox = (*MacOSSandbox)(nil)

// filepath 引用（避免误删 import）。
var _ = filepath.Join
```

//go:build linux

package sandbox

import (
	"fmt"
	"os"
	"os/exec"

	landlock "github.com/landlock-lsm/go-landlock"
)

// LinuxSandbox 用 Landlock（文件系统）+ unshare -n（网络）限制命令。
//
// Landlock 是内核 5.13+ 的纯 syscall 沙箱（无外部依赖、非特权）。
// 网络隔离：内核 6.7+ 有 Landlock 网络规则；低版本用 unshare -n
// （空网络命名空间，user namespaces 可用时非 root 可执行）。
type LinuxSandbox struct {
	allowNetwork bool
	workDir      string
}

// newLinuxSandbox 创建 Linux 沙箱。
func newLinuxSandbox(cfg Config) *LinuxSandbox {
	return &LinuxSandbox{
		allowNetwork: cfg.AllowNetwork,
		workDir:      cfg.WorkDir,
	}
}

// Wrap 应用 Landlock 文件系统规则 + unshare -n 网络隔离。
func (s *LinuxSandbox) Wrap(cmd *exec.Cmd) *exec.Cmd {
	// ── 文件系统隔离（Landlock） ──
	// go-landlock 在命令启动前（exec 内部）应用规则。
	// 工作目录可读写，系统目录只读。
	rl := landlock.New(
		landlock.ReadOnly("/usr", "/etc", "/bin", "/sbin", "/lib", "/opt"),
		landlock.ReadWrite(s.workDir),
	)
	rl = rl.RestrictPaths()

	// 通过 cmd.SysProcAttr 或环境变量传递？go-landlock 需要在 exec 前应用。
	// 方式：将 RestrictPaths 生成的限制作为前置动作绑定到 cmd。
	// 见 landlock 库文档：BestEffort() 或 RestrictPaths() 返回 Ruleset，
	// 需要在自己 fork/exec 时应用。对 exec.Cmd 的适配：
	//   1. 用 cmd.Start 前调用 rl 的 Apply（通过自定义 Cmd 包装）
	//   2. 或改用 landlock 提供的 exec.Cmd 兼容方案
	//
	// 实现细节：go-landlock 提供 landlock.RestrictPaths() 后，需在
	// cmd.Start() 前调用。ShellTool 只调 cmd.CombinedOutput()——因此
	// LinuxSandbox 用以下方案：预检查 + 在 cmd 上加钩子不现实，
	// 改为在 Wrap 时验证并降级（若 landlock 不可用 → 警告 + 返回原 cmd）。

	// ── 网络隔离（unshare -n，非 Landlock 网络规则时） ──
	if !s.allowNetwork {
		// unshare -n 切到空网络命名空间（无网络接口）。
		// 非 root 依赖 user namespaces（Linux 3.8+，多数发行版默认开启）。
		inner := append([]string{"-n"}, cmd.Args...)
		cmd.Args = append([]string{"unshare"}, inner...)
		cmd.Path = "unshare"
	}
	return cmd
}

// AllowsNetwork 报告是否允许网络。
func (s *LinuxSandbox) AllowsNetwork() bool { return s.allowNetwork }

// Close 无资源需释放。
func (s *LinuxSandbox) Close() error { return nil }

// 确保沙箱在 linux 下可用（编译期断言）。
var _ Sandbox = (*LinuxSandbox)(nil)

var (
	_ = fmt.Sprintf
	_ = os.Getwd
)
```

// CheckPermission 检查 shell 命令是否需要用户确认。
//
// 策略：
//   - 沙箱开启且请求 network=true → 需确认（网络访问是风险操作）
//   - 包含危险命令模式（rm -rf、sudo、chmod 777 等）→ 需确认
//   - 其他命令 → 直接允许
func (t *ShellTool) CheckPermission(args string) PermissionResult {
	if t.sandbox != nil {
		var params struct {
			Network bool `json:"network"`
		}
		if err := parseArgs(args, &params); err == nil && params.Network {
			return PermissionResult{
				Allow:  false,
				Reason: "需要网络访问，请确认",
			}
		}
	}
	if isDangerousShellCommand(args) {
		return PermissionResult{
			Allow:  false,
			Reason: "该命令可能有风险，需要确认执行",
		}
	}
	return PermissionResult{Allow: true}
}

package tool

import (
	"os/exec"
	"testing"
)

// mockSandbox 用于测试 ShellTool 的 Wrap 调用。
type mockSandbox struct{ wrapped bool }

func (m *mockSandbox) Wrap(cmd *exec.Cmd) *exec.Cmd { m.wrapped = true; return cmd }
func (m *mockSandbox) AllowsNetwork() bool          { return false }
func (m *mockSandbox) Close() error                 { return nil }

func TestShellTool_WithSandbox_Wraps(t *testing.T) {
	sb := &mockSandbox{}
	st := NewShellToolWithSandbox(sb)

	// 执行一个无害命令。
	_, err := st.Execute(`{"command": "echo hi"}`)
	if err != nil {
		t.Fatalf("unexpected error: %v", err)
	}
	if !sb.wrapped {
		t.Error("ShellTool should wrap cmd when sandbox is set")
	}
}

func TestShellTool_NetworkPermission(t *testing.T) {
	st := NewShellToolWithSandbox(&mockSandbox{})

	// network=true → 需确认。
	perm := st.CheckPermission(`{"command": "curl https://x.com", "network": true}`)
	if perm.Allow {
		t.Error("network:true should require confirmation")
	}
	if !strings.Contains(perm.Reason, "网络") {
		t.Errorf("reason should mention network, got %q", perm.Reason)
	}

	// 默认（无 network）→ 允许（非危险命令）。
	perm = st.CheckPermission(`{"command": "ls -la"}`)
	if !perm.Allow {
		t.Error("ls should be allowed by default")
	}

	// 无沙箱时 network 参数忽略 → 允许。
	stPlain := NewShellTool()
	perm = stPlain.CheckPermission(`{"command": "curl https://x.com", "network": true}`)
	if !perm.Allow {
		t.Error("network param should be ignored without sandbox")
	}
}
```
